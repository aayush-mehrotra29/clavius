cmake_minimum_required(VERSION 3.18)
project(ClaviusChronoReference LANGUAGES CXX)
find_package(Chrono COMPONENTS Vehicle CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD ${CHRONO_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(clavius_chrono_reference reference.cpp)
target_include_directories(clavius_chrono_reference PRIVATE ${CHRONO_INCLUDE_DIRS})
target_compile_options(clavius_chrono_reference PRIVATE ${CHRONO_CXX_FLAGS})
target_link_libraries(clavius_chrono_reference ${CHRONO_LIBRARIES})
set_target_properties(clavius_chrono_reference PROPERTIES BUILD_RPATH "${CHRONO_LIBRARY_DIRS}")
