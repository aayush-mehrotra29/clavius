cmake_minimum_required(VERSION 3.18)
project(ClaviusChronoRoverProbe LANGUAGES CXX)
find_package(Chrono COMPONENTS Vehicle CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD ${CHRONO_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(clavius_viper_rigid_probe viper-rigid.cpp)
target_include_directories(clavius_viper_rigid_probe PRIVATE ${CHRONO_INCLUDE_DIRS})
target_compile_options(clavius_viper_rigid_probe PRIVATE ${CHRONO_CXX_FLAGS})
target_link_libraries(clavius_viper_rigid_probe ${CHRONO_LIBRARIES})
set_target_properties(clavius_viper_rigid_probe PROPERTIES BUILD_RPATH "${CHRONO_LIBRARY_DIRS}")
