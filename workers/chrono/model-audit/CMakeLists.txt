cmake_minimum_required(VERSION 3.18)
project(ClaviusChronoModelAudit LANGUAGES CXX)
find_package(Chrono COMPONENTS Vehicle CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD ${CHRONO_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(clavius_viper_audit viper.cpp)
target_include_directories(clavius_viper_audit PRIVATE ${CHRONO_INCLUDE_DIRS})
target_compile_options(clavius_viper_audit PRIVATE ${CHRONO_CXX_FLAGS})
target_link_libraries(clavius_viper_audit ${CHRONO_LIBRARIES})
set_target_properties(clavius_viper_audit PROPERTIES BUILD_RPATH "${CHRONO_LIBRARY_DIRS}")
add_executable(clavius_viper_wheel_mass_frames wheel-mass-frames.cpp)
target_include_directories(clavius_viper_wheel_mass_frames PRIVATE ${CHRONO_INCLUDE_DIRS})
target_compile_options(clavius_viper_wheel_mass_frames PRIVATE ${CHRONO_CXX_FLAGS})
target_link_libraries(clavius_viper_wheel_mass_frames ${CHRONO_LIBRARIES})
set_target_properties(clavius_viper_wheel_mass_frames PROPERTIES BUILD_RPATH "${CHRONO_LIBRARY_DIRS}")
