cmake_minimum_required(VERSION 3.18)
project(ClaviusViperFrameRepair LANGUAGES CXX)
find_package(Chrono CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD ${CHRONO_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CLAVIUS_VIPER_SOURCE_DIR)
  message(FATAL_ERROR "Build through scripts/build_viper_adapted.py to prepare verified source")
endif()
set(MODEL_DIR "${CLAVIUS_VIPER_SOURCE_DIR}/src/chrono_models/robot/viper")
add_library(clavius_viper_repaired STATIC "${MODEL_DIR}/Viper.cpp")
target_include_directories(clavius_viper_repaired PUBLIC "${MODEL_DIR}" ${CHRONO_INCLUDE_DIRS})
target_compile_options(clavius_viper_repaired PRIVATE ${CHRONO_CXX_FLAGS})
target_link_libraries(clavius_viper_repaired PUBLIC ${CHRONO_LIBRARIES})
foreach(target IN ITEMS clavius_viper_frames_repaired clavius_viper_frames_original)
  add_executable(${target} frames.cpp)
  target_include_directories(${target} PRIVATE ${CHRONO_INCLUDE_DIRS})
  target_compile_options(${target} PRIVATE ${CHRONO_CXX_FLAGS})
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${CHRONO_LIBRARY_DIRS}")
endforeach()
target_link_libraries(clavius_viper_frames_repaired PRIVATE clavius_viper_repaired)
target_compile_definitions(clavius_viper_frames_original PRIVATE CLAVIUS_ORIGINAL=1)
target_link_libraries(clavius_viper_frames_original PRIVATE ${CHRONO_LIBRARIES})
add_executable(clavius_viper_rigid_repaired "${CLAVIUS_VIPER_SOURCE_DIR}/viper-rigid.cpp")
target_compile_options(clavius_viper_rigid_repaired PRIVATE ${CHRONO_CXX_FLAGS})
target_link_libraries(clavius_viper_rigid_repaired PRIVATE clavius_viper_repaired)
set_target_properties(clavius_viper_rigid_repaired PROPERTIES BUILD_RPATH "${CHRONO_LIBRARY_DIRS}")
