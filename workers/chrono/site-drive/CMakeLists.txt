cmake_minimum_required(VERSION 3.18)
project(ClaviusSiteDrive LANGUAGES CXX)
find_package(Chrono CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD ${CHRONO_CXX_STANDARD})
add_executable(clavius_site_drive main.cpp)
target_include_directories(clavius_site_drive PRIVATE "${ADAPTED_SOURCE}/src/chrono_models/robot/viper" ${CHRONO_INCLUDE_DIRS})
target_compile_options(clavius_site_drive PRIVATE ${CHRONO_CXX_FLAGS})
target_link_libraries(clavius_site_drive PRIVATE "${ADAPTED_LIBRARY}" ${CHRONO_LIBRARIES})
set_target_properties(clavius_site_drive PROPERTIES BUILD_RPATH "${CHRONO_LIBRARY_DIRS}")
